// ─── Energy Slider ─────────────────────────────────────────────────────────────
function EnergySlider({ label, value, onChange, color }) {
  return (
    <div style={{ marginBottom: 14 }}>
      <div style={{ display: "flex", justifyContent: "space-between", marginBottom: 6 }}>
        <span style={{ color: COLORS.muted, fontSize: 12, fontFamily: "'DM Sans', sans-serif" }}>{label}</span>
        <span style={{ color: color, fontSize: 12, fontWeight: 600 }}>{value}%</span>
      </div>
      <div style={{ position: "relative", height: 6, borderRadius: 3, background: "rgba(255,255,255,0.08)", cursor: "pointer" }}
        onClick={e => {
          const rect = e.currentTarget.getBoundingClientRect();
          const pct = Math.round(Math.max(0, Math.min(100, ((e.clientX - rect.left) / rect.width) * 100)));
          onChange(pct);
        }}
      >
        <div style={{ position: "absolute", left: 0, top: 0, height: "100%", width: `${value}%`, borderRadius: 3, background: `linear-gradient(90deg, ${color}88, ${color})`, transition: "width 0.2s" }}/>
        <div style={{ position: "absolute", top: "50%", left: `${value}%`, transform: "translate(-50%, -50%)", width: 14, height: 14, borderRadius: "50%", background: color, boxShadow: `0 0 8px ${color}`, border: "2px solid rgba(255,255,255,0.3)" }}/>
      </div>
    </div>
  );
}

// ─── NOVA SCREEN ──────────────────────────────────────────────────────────────
function NovaScreen() {
  const [messages, setMessages] = useState([
    { from: "nova", text: "I noticed you've been feeling overwhelmed this week. I'm here for you. What would you like to do?" }
  ]);
  const [input, setInput] = useState("");
  const [loading, setLoading] = useState(false);
  const bottomRef = useRef(null);

  const actions = ["Reflect on my day", "Reframe my thoughts", "Breathing exercise", "Just talk"];

  const sendMessage = async (text) => {
    if (!text.trim() || loading) return;
    const userMsg = { from: "user", text };
    const newMsgs = [...messages, userMsg];
    setMessages(newMsgs);
    setInput("");
    setLoading(true);

    try {
      const res = await fetch("https://api.anthropic.com/v1/messages", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          model: "claude-sonnet-4-20250514",
          max_tokens: 1000,
          system: `You are Nova, Zenara's compassionate AI mental health companion. You are warm, empathetic, and supportive. You help users reflect on their emotions, reframe negative thoughts, and practice coping strategies. You are NOT a therapist or medical professional — always remind users to speak with their therapist (Dr. Hayes) for clinical support. Keep responses concise (2-3 sentences), warm, and conversational. Use gentle, supportive language. If you detect any self-harm language, immediately provide crisis resources.`,
          messages: newMsgs.map(m => ({ role: m.from === "user" ? "user" : "assistant", content: m.text }))
        })
      });
      const data = await res.json();
      const reply = data.content?.[0]?.text || "I'm here with you. Tell me more.";
      setMessages(prev => [...prev, { from: "nova", text: reply }]);
    } catch {
      setMessages(prev => [...prev, { from: "nova", text: "I'm having a little trouble connecting right now. Take a breath — I'm still here with you. 💜" }]);
    }
    setLoading(false);
  };

  useEffect(() => { bottomRef.current?.scrollIntoView({ behavior: "smooth" }); }, [messages, loading]);

  return (
    <div style={{ display: "flex", flexDirection: "column", height: "100%", overflow: "hidden" }}>
      {/* Header */}
      <div style={{ padding: "12px 20px 16px", textAlign: "center", flexShrink: 0 }}>
        <h2 style={{ color: COLORS.white, fontSize: 22, margin: "0 0 2px", fontFamily: "'Playfair Display', serif" }}>Nova</h2>
        <p style={{ color: COLORS.muted, fontSize: 12, margin: 0, fontFamily: "'DM Sans', sans-serif" }}>AI Companion</p>
      </div>

      {/* Nova Moon Avatar */}
      <div style={{ textAlign: "center", marginBottom: 16, flexShrink: 0 }}>
        <div style={{
          width: 120, height: 120, borderRadius: "50%", margin: "0 auto",
          background: "radial-gradient(circle at 35% 35%, #3d2a6e, #1a1040)",
          boxShadow: "0 0 40px rgba(124,111,247,0.4), 0 0 80px rgba(124,111,247,0.15)",
          display: "flex", alignItems: "center", justifyContent: "center", fontSize: 52,
          border: "2px solid rgba(168,154,247,0.3)"
        }}>
          🌙
        </div>
      </div>

      {/* Messages */}
      <div style={{ flex: 1, overflowY: "auto", padding: "0 20px", scrollbarWidth: "none" }}>
        {messages.map((msg, i) => (
          <div key={i} style={{ marginBottom: 12, display: "flex", justifyContent: msg.from === "user" ? "flex-end" : "flex-start" }}>
            <div style={{
              maxWidth: "80%", padding: "12px 16px", borderRadius: msg.from === "user" ? "18px 18px 4px 18px" : "18px 18px 18px 4px",
              background: msg.from === "user" ? `linear-gradient(135deg, ${COLORS.purple}, ${COLORS.purpleDark})` : COLORS.bgElevated,
              border: msg.from === "nova" ? `1px solid ${COLORS.border}` : "none",
              color: COLORS.white, fontSize: 13, lineHeight: 1.5, fontFamily: "'DM Sans', sans-serif"
            }}>
              {msg.text}
            </div>
          </div>
        ))}
        {loading && (
          <div style={{ display: "flex", gap: 4, padding: "12px 16px", background: COLORS.bgElevated, borderRadius: "18px 18px 18px 4px", width: "fit-content", marginBottom: 12, border: `1px solid ${COLORS.border}` }}>
            {[0,1,2].map(i => (
              <div key={i} style={{ width: 6, height: 6, borderRadius: "50%", background: COLORS.purpleLight, animation: `pulse 1.2s ${i * 0.2}s infinite` }}/>
            ))}
          </div>
        )}
        <div ref={bottomRef}/>
      </div>

      {/* Quick actions */}
      {messages.length === 1 && (
        <div style={{ padding: "8px 20px", display: "flex", flexDirection: "column", gap: 8, flexShrink: 0 }}>
          {actions.map(a => (
            <button key={a} onClick={() => sendMessage(a)} style={{
              background: COLORS.bgElevated, border: `1px solid ${COLORS.border}`,
              borderRadius: 14, padding: "12px 16px", color: COLORS.white, fontSize: 13,
              cursor: "pointer", textAlign: "left", fontFamily: "'DM Sans', sans-serif",
              display: "flex", alignItems: "center", gap: 10
            }}>
              <span style={{ color: COLORS.purple }}>→</span> {a}
            </button>
          ))}
        </div>
      )}
